def analyze_token(token):
    score = 0

    market_cap = token.get("marketCap", 0)
    liquidity = token.get("liquidity", 0)
    volume = token.get("volume", 0)
    change = token.get("priceChange", 0)

    if market_cap > 1_000_000:
        score += 25

    if liquidity > 100_000:
        score += 25

    if volume > 500_000:
        score += 25

    if change > 0:
        score += 25

    if score >= 75:
        signal = "BUY ✅"
        risk = "LOW"
    elif score >= 50:
        signal = "WATCH 🟡"
        risk = "MEDIUM"
    else:
        signal = "AVOID ❌"
        risk = "HIGH"

    return score, signal, risk
    
def get_token_data(contract):
    url = f"https://api.dexscreener.com/latest/dex/tokens/{contract}"

    try:
        response = requests.get(url, timeout=10)

        if response.status_code != 200:
            return None

        data = response.json()

        if "pairs" not in data or len(data["pairs"]) == 0:
            return None

        pair = data["pairs"][0]

        return {
            "name": pair["baseToken"]["name"],
            "symbol": pair["baseToken"]["symbol"],
            "price": pair.get("priceUsd", "Unknown"),
            "marketCap": pair.get("marketCap", "Unknown"),
            "liquidity": pair["liquidity"]["usd"],
            "volume": pair["volume"]["h24"],
            "priceChange": pair["priceChange"]["h24"],
            "fdv": pair.get("fdv", "Unknown"),
            "url": pair["url"],
        }

    except Exception:
        return None
def get_trending_tokens():
    url = "https://api.dexscreener.com/token-profiles/latest/v1"

    try:
        response = requests.get(url, timeout=10)

        if response.status_code != 200:
            return None

        data = response.json()

        if not isinstance(data, list):
            return None

        trending = []

        for token in data[:5]:
            trending.append({
                "name": token.get("tokenName", "Unknown"),
                "symbol": token.get("tokenSymbol", ""),
                "chain": token.get("chainId", ""),
                "url": token.get("url", "")
            })

        return trending

    except Exception:
        return None

def calculate_ai_score(token):
    score = 50

    try:
        liquidity = float(token["liquidity"])
        volume = float(token["volume"])
        market_cap = float(token["marketCap"])
        change = float(token["priceChange"])
    except Exception:
        return 50, "🟡 WATCH", "🟡 MEDIUM"

    if liquidity >= 500000:
        score += 15
    elif liquidity >= 100000:
        score += 10
    elif liquidity >= 50000:
        score += 5

    if volume >= 1000000:
        score += 15
    elif volume >= 250000:
        score += 10
    elif volume >= 50000:
        score += 5

    if market_cap >= 10000000:
        score += 10
    elif market_cap >= 1000000:
        score += 5

    if change >= 20:
        score += 10
    elif change >= 5:
        score += 5
    elif change <= -20:
        score -= 10

    score = max(0, min(score, 100))

    if score >= 85:
        signal = "🟢 STRONG BUY"
        risk = "🟢 LOW"
    elif score >= 70:
        signal = "🟢 BUY"
        risk = "🟡 MEDIUM"
    elif score >= 50:
        signal = "🟡 WATCH"
        risk = "🟡 MEDIUM"
    else:
        signal = "🔴 AVOID"
        risk = "🔴 HIGH"

    return score, signal, risk
